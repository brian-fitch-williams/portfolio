110CTAlignment CTVolume::EstimateModelOffset(
111 const std::vector<std::array<double, 3>>& lumenSamples) const
112{
113 CTAlignment result;
114 if (!IsLoaded() || lumenSamples.empty())
115 {
116 return result;
117 }
118
119 vtkDataArray* scalars = m_Image->GetPointData()->GetScalars();
120 if (scalars == nullptr)
121 {
122 return result;
123 }
124
125 int dim[3];
126 double origin[3];
127 double spacing[3];
128 m_Image->GetDimensions(dim);
129 m_Image->GetOrigin(origin);
130 m_Image->GetSpacing(spacing);
131
132 // One pass to a flat mask, so scoring a shift is plain indexing rather than
133 // a virtual call per sample.
134 const std::size_t total = static_cast<std::size_t>(dim[0]) *
135 static_cast<std::size_t>(dim[1]) * static_cast<std::size_t>(dim[2]);
136 std::vector<std::uint8_t> air(total, 0);
137 bool anyAir = false;
138 for (std::size_t i = 0; i < total; ++i)
139 {
140 if (scalars->GetTuple1(static_cast<vtkIdType>(i)) <= AirThreshold)
141 {
142 air[i] = 1;
143 anyAir = true;
144 }
145 }
146 if (!anyAir)
147 {
148 return result;
149 }
150
151 const std::size_t minInBounds = static_cast<std::size_t>(lumenSamples.size() * 0.9);
152 auto Score = [&](const std::array<double, 3>& shift) -> double
153 {
154 std::size_t inBounds = 0;
155 std::size_t hits = 0;
156 for (const auto& p : lumenSamples)
157 {
158 const int i = static_cast<int>(std::lround((p[0] + shift[0] - origin[0]) / spacing[0]));
159 const int j = static_cast<int>(std::lround((p[1] + shift[1] - origin[1]) / spacing[1]));
160 const int k = static_cast<int>(std::lround((p[2] + shift[2] - origin[2]) / spacing[2]));
161 if (i < 0 || i >= dim[0] || j < 0 || j >= dim[1] || k < 0 || k >= dim[2])
162 {
163 continue;
164 }
165 ++inBounds;
166 const std::size_t index =
167 (static_cast<std::size_t>(k) * dim[1] + j) * dim[0] + i;
168 hits += air[index];
169 }
170 if (inBounds < minInBounds)
171 {
172 return -1.0;
173 }
174 return static_cast<double>(hits) / static_cast<double>(inBounds);
175 };
176
177 double span[3];
178 for (int a = 0; a < 3; ++a)
179 {
180 span[a] = (dim[a] - 1) * spacing[a];
181 }
182
183 // Coarse sweep of the whole volume, then two refinements around the winner.
184 // Each pass is a regular grid centred on the current best, so the origin
185 // (and later the coarse winner) is always a sample.
186 result.valid = true;
187 result.overlap = Score(result.offset);
188
189 const double steps[3] = { 8.0, 2.0, 1.0 };
190 for (int pass = 0; pass < 3; ++pass)
191 {
192 const double step = steps[pass];
193 const std::array<double, 3> centre = result.offset;
194 int reach[3];
195 for (int a = 0; a < 3; ++a)
196 {
197 const double mm = (pass == 0) ? span[a] / 2.0 : step * 4.0;
198 reach[a] = std::max(1, static_cast<int>(std::ceil(mm / step)));
199 }
200
201 for (int ix = -reach[0]; ix <= reach[0]; ++ix)
202 {
203 for (int iy = -reach[1]; iy <= reach[1]; ++iy)
204 {
205 for (int iz = -reach[2]; iz <= reach[2]; ++iz)
206 {
207 const std::array<double, 3> shift{
208 centre[0] + ix * step,
209 centre[1] + iy * step,
210 centre[2] + iz * step
211 };
212 const double value = Score(shift);
213 if (value > result.overlap)
214 {
215 result.overlap = value;
216 result.offset = shift;
217 }
218 }
219 }
220 }
221 }
222
223 return result;
224}
