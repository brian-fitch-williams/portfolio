530 auto GravityUp = [](const double fwd[3], const double prevUp[3], double out[3])

534 vtkMath::Cross(fwd, worldZ, right);

540 SubtractTangent(level, fwd);
541 if (!Normalize3(level))
542 {
543 return false;
544 }
545 if (vtkMath::Dot(level, prevUp) < 0.0)
546 {
547 level[0] = -level[0];
548 level[1] = -level[1];
549 level[2] = -level[2];

555 auto RollToward = [](double up[3], const double fwd[3], const double desired[3], double maxRad)
556 {
557 double a[3] = { up[0], up[1], up[2] };

559 if (!Normalize3(a))
560 {
561 return;
562 }

565 if (!Normalize3(b))
566 {
567 return;
568 }

571 vtkMath::Cross(a, b, cross);
572 const double err = std::atan2(vtkMath::Dot(fwd, cross), vtkMath::Dot(a, b));

574 {
575 Copy3(a, up);
576 return;
577 }
578 const double step = std::max(-maxRad, std::min(maxRad, err));
579 RotateAbout(a, fwd, step);
580 SubtractTangent(a, fwd);
581 if (Normalize3(a))
582 {
583 Copy3(a, up);
584 }
585 };
586
587 double prevFwd[3] = { 0.0, 0.0, -1.0 };
588 ForwardOf(track.front(), prevFwd);
589
590 double up[3] = { 0.0, 1.0, 0.0 };
591 double gravity[3];

594 Copy3(gravity, up);
595 }

599 if (!Normalize3(up))

706 const float* in,
707 float* out,
708 int n,
709 double spacing,
710 std::vector<int>& hullIndex,
711 std::vector<double>& hullBound)
712{
713 if (n <= 0)
714 {
715 return;
716 }
717
718 const double sp2 = spacing * spacing;
719 int k = 0;
720 hullIndex[0] = 0;
721 hullBound[0] = -1.0e300;
722 hullBound[1] = 1.0e300;
723
724 for (int q = 1; q < n; ++q)
725 {
726 double crossing = 0.0;
727 while (true)
728 {
729 const int p = hullIndex[k];
730 crossing = ((in[q] + sp2 * q * q) - (in[p] + sp2 * p * p)) / (2.0 * sp2 * (q - p));
731 if (k == 0 || crossing > hullBound[k])
732 {
733 break;
734 }
735 --k;
736 }
737 ++k;
738 hullIndex[k] = q;
739 hullBound[k] = crossing;
740 hullBound[k + 1] = 1.0e300;
741 }
742
743 k = 0;
744 for (int q = 0; q < n; ++q)
745 {
746 while (hullBound[k + 1] < q)
747 {
748 ++k;
749 }
750 const double offset = spacing * (q - hullIndex[k]);
751 out[q] = static_cast<float>(in[hullIndex[k]] + offset * offset);
752 }
753}

843 {
844 extent[a] = bounds[2 * a + 1] - bounds[2 * a];
845 if (extent[a] <= 0.0)
846 {
847 return nullptr;
848 }
849 }
850
851 // Third-generation bronchi are only a few millimetres wide, so the grid
852 // has to stay fine. The budget coarsens it for oversized models instead
853 // of exhausting memory.
854 constexpr double kFinestSpacing = 0.7;
855 constexpr double kVoxelBudget = 1.4e7;
856 const double padded = (extent[0] + 4.0) * (extent[1] + 4.0) * (extent[2] + 4.0);
857 const double spacing = std::max(kFinestSpacing, std::cbrt(padded / kVoxelBudget));
858
859 auto field = std::make_unique<LumenField>();
860 field->spacing = spacing;
861 for (int a = 0; a < 3; ++a)
862 {
863 field->origin[a] = bounds[2 * a] - 2.0 * spacing;
864 field->dim[a] = static_cast<int>(std::ceil((extent[a] + 4.0 * spacing) / spacing)) + 1;
865 }
866
867 const std::size_t total = static_cast<std::size_t>(field->dim[0]) *
868 static_cast<std::size_t>(field->dim[1]) *
869 static_cast<std::size_t>(field->dim[2]);
870
871 // Voxelise each shell on its own and union the results. Rasterising the
872 // appended mesh would let overlapping segments (trachea into bronchi)
873 // cancel one another under the even-odd rule.
874 std::vector<std::uint8_t> inside(total, 0);
875 int wholeExtent[6] = { 0, field->dim[0] - 1, 0, field->dim[1] - 1, 0, field->dim[2] - 1 };
876 for (const auto& shell : shells)
877 {
878 if (!shell || shell->GetNumberOfPolys() == 0)
879 {
880 continue;
881 }
882
883 auto stencil = vtkSmartPointer<vtkPolyDataToImageStencil>::New();
884 stencil->SetInputData(shell);
885 stencil->SetOutputOrigin(field->origin);
886 stencil->SetOutputSpacing(spacing, spacing, spacing);
887 stencil->SetOutputWholeExtent(wholeExtent);
888 stencil->SetTolerance(0.0);
889 stencil->Update();
890 if (pulse)
891 {
892 pulse();
893 }
894
895 vtkImageStencilData* filled = stencil->GetOutput();
896 if (!filled)
897 {
898 continue;
899 }
900
901 for (int k = 0; k < field->dim[2]; ++k)
902 {
903 if (pulse && (k % 16) == 0)
904 {
905 pulse();
906 }
907 for (int j = 0; j < field->dim[1]; ++j)
908 {
909 const std::size_t row = field->VoxelIndex(0, j, k);
910 int iter = 0;
911 int r1 = 0;
912 int r2 = 0;
913 while (filled->GetNextExtent(r1, r2, 0, field->dim[0] - 1, j, k, iter))
914 {
915 for (int i = r1; i <= r2; ++i)
916 {
917 inside[row + static_cast<std::size_t>(i)] = 1;
918 }
919 }
920 }
921 }
922 }
923
924 std::vector<float> squared(total);
925 for (std::size_t i = 0; i < total; ++i)
926 {
927 squared[i] = inside[i] ? kEdtInfinity : 0.0f;
928 }
929 SquaredEdt3D(squared, field->dim, spacing);
930
931 // The penalty saturates once the lumen is wide, so the trachea stays
932 // cheap to cross while narrow branches pull the route onto their axis.
933 constexpr double kOpenRadius = 8.0;
934 field->nodeAt.assign(total, -1);
935 for (std::size_t i = 0; i < total; ++i)
936 {
937 if (!inside[i])
938 {
939 continue;
940 }
941 const double wall = std::sqrt(static_cast<double>(squared[i]));
942 field->nodeAt[i] = static_cast<std::int32_t>(field->nodeVox.size());
943 field->nodeVox.push_back(static_cast<std::int32_t>(i));
944 field->wallDist.push_back(static_cast<float>(wall));
945 const double ratio = kOpenRadius / std::min(kOpenRadius, std::max(0.35, wall));
946 field->stepCost.push_back(static_cast<float>(ratio * ratio * ratio));
947 }
948
949 constexpr std::size_t kMinLumenVoxels = 256;
950 if (field->nodeVox.size() < kMinLumenVoxels)
951 {
952 return nullptr;
953 }
954 return field;
955}
956
962void SmoothCentersInLumen(
963 const LumenField& field,
964 std::vector<std::array<double, 3>>& centers,
965 int passes)
966{
967 if (centers.size() < 3)
968 {
969 return;
970 }
971
972 // Measured once, so repeated passes cannot compound small losses of
973 // clearance into a drift toward the wall.
974 std::vector<double> floorDist(centers.size());
975 for (std::size_t k = 0; k < centers.size(); ++k)
976 {
977 floorDist[k] = 0.8 * field.WallDistAt(centers[k].data());
978 }
979
980 for (int pass = 0; pass < passes; ++pass)
981 {
982 for (std::size_t k = 1; k + 1 < centers.size(); ++k)
983 {
984 std::array<double, 3> moved{};
985 for (int a = 0; a < 3; ++a)
986 {
987 moved[a] = 0.25 * centers[k - 1][a] +
988 0.50 * centers[k][a] +
989 0.25 * centers[k + 1][a];
990 }
991 if (field.WallDistAt(moved.data()) >= floorDist[k])
992 {
993 centers[k] = moved;
994 }
995 }
996 }
997}

1228bool BronchoEngine::ComputeLumenCenters(
1229 const double start_pos[3],
1230 const double target_pos[3],
1231 std::vector<std::array<double, 3>>& centers) const
1232{
1233 if (!m_Lumen || !m_Lumen->IsValid())
1234 {
1235 return false;
1236 }
1237 const LumenField& field = *m_Lumen;
1238
1239 // Generous enough to recover a camera that has drifted through a wall,
1240 // tight enough that a click on empty space is rejected.
1241 constexpr double kSnapRadius = 18.0;
1242 const std::int32_t startNode = field.FindNode(start_pos, kSnapRadius);
1243 const std::int32_t targetNode = field.FindNode(target_pos, kSnapRadius);
1244 if (startNode < 0 || targetNode < 0)
1245 {
1246 return false;
1247 }
1248
1249 if (startNode == targetNode)
1250 {
1251 double only[3];
1252 field.VoxelCenter(startNode, only);
1253 centers.assign(1, { only[0], only[1], only[2] });
1254 return true;
1255 }
1256
1257 const std::size_t nodeCount = field.nodeVox.size();
1258 std::vector<float> cost(nodeCount, std::numeric_limits<float>::max());
1259 std::vector<std::int32_t> came(nodeCount, -1);
1260
1261 using Entry = std::pair<float, std::int32_t>;
1262 std::priority_queue<Entry, std::vector<Entry>, std::greater<Entry>> frontier;
1263 cost[startNode] = 0.0f;
1264 frontier.emplace(0.0f, startNode);
1265
1266 const int nx = field.dim[0];
1267 const int ny = field.dim[1];
1268 const int nz = field.dim[2];
1269 bool reached = false;
1270
1271 while (!frontier.empty())
1272 {
1273 const Entry current = frontier.top();
1274 frontier.pop();
1275 const std::int32_t node = current.second;
1276 if (current.first > cost[node])
1277 {
1278 continue;
1279 }
1280 if (node == targetNode)
1281 {
1282 reached = true;
1283 break;
1284 }
1285
1286 const std::int32_t vox = field.nodeVox[node];
1287 const int i = vox % nx;
1288 const int j = (vox / nx) % ny;
1289 const int k = vox / (nx * ny);
1290
1291 for (int dk = -1; dk <= 1; ++dk)
1292 {
1293 for (int dj = -1; dj <= 1; ++dj)
1294 {
1295 for (int di = -1; di <= 1; ++di)
1296 {
1297 if (di == 0 && dj == 0 && dk == 0)
1298 {
1299 continue;
1300 }
1301 const int ni = i + di;
1302 const int nj = j + dj;
1303 const int nk = k + dk;
1304 if (ni < 0 || ni >= nx || nj < 0 || nj >= ny || nk < 0 || nk >= nz)
1305 {
1306 continue;
1307 }
1308 const std::int32_t next = field.nodeAt[field.VoxelIndex(ni, nj, nk)];
1309 if (next < 0)
1310 {
1311 continue;
1312 }
1313
1314 const double stepLen = field.spacing *
1315 std::sqrt(static_cast<double>(di * di + dj * dj + dk * dk));
1316 const float candidate = cost[node] +
1317 static_cast<float>(stepLen * field.stepCost[next]);
1318 if (candidate < cost[next])
1319 {
1320 cost[next] = candidate;
1321 came[next] = node;
1322 frontier.emplace(candidate, next);
1323 }
1324 }
1325 }
1326 }
1327 }
1328
1329 if (!reached)
1330 {
1331 return false;
1332 }
1333
1334 std::vector<std::int32_t> backward;
1335 for (std::int32_t node = targetNode; node >= 0; node = came[node])
1336 {
1337 backward.push_back(node);
1338 }
1339
1340 centers.clear();
1341 centers.reserve(backward.size());
1342 for (auto it = backward.rbegin(); it != backward.rend(); ++it)
1343 {
1344 double p[3];
1345 field.VoxelCenter(*it, p);
1346 centers.push_back({ p[0], p[1], p[2] });
1347 }
1348
1349 SmoothCentersInLumen(field, centers, 24);
1350 return centers.size() >= 2;
1351}

1356std::vector<std::array<double, 3>> BronchoEngine::ComputeSurfaceCenters(
1357 const double start_pos[3],
1358 const double target_pos[3]) const
1359{
1360 std::vector<std::array<double, 3>> centers;
1361
1362 if (!m_PointLocator || !m_CellLocator || !m_NavMesh)
1363 {
1364 return centers;
1365 }
1366
1367 vtkIdType start_id = m_PointLocator->FindClosestPoint(start_pos);
1368 vtkIdType target_id = m_PointLocator->FindClosestPoint(target_pos);
1369
1370 if (start_id < 0 || target_id < 0) return centers;
1371
1372 auto dijkstra = vtkSmartPointer<vtkDijkstraGraphGeodesicPath>::New();
1373 dijkstra->SetInputData(m_NavMesh);
1374 dijkstra->SetStartVertex(start_id);
1375 dijkstra->SetEndVertex(target_id);
1376 dijkstra->StopWhenEndReachedOn();
1377 dijkstra->Update();
1378
1379 auto pathIds = dijkstra->GetIdList();
1380 if (!pathIds || pathIds->GetNumberOfIds() == 0) return centers;
1381
1382 const vtkIdType num_ids = pathIds->GetNumberOfIds();
1383 std::vector<vtkIdType> ordered;
1384 ordered.reserve(static_cast<size_t>(num_ids));
1385 for (vtkIdType i = 0; i < num_ids; ++i)
1386 {
1387 ordered.push_back(pathIds->GetId(i));
1388 }
1389 if (!ordered.empty() && ordered.front() == target_id && ordered.back() == start_id)
1390 {
1391 std::reverse(ordered.begin(), ordered.end());
1392 }
1393 else if (!ordered.empty() && ordered.back() == start_id && ordered.front() != start_id)
1394 {
1395 std::reverse(ordered.begin(), ordered.end());
1396 }
1397
1398 auto SampleTangent = [&](size_t n, double tangent[3])
1399 {
1400 const size_t i0 = (n > 3) ? (n - 3) : 0;
1401 const size_t i1 = std::min(ordered.size() - 1, n + 3);
1402 tangent[0] = 0.0;
1403 tangent[1] = 0.0;
1404 tangent[2] = 1.0;
1405 if (i0 == i1)
1406 {
1407 return;
1408 }
1409 double a[3], b[3];
1410 m_NavMesh->GetPoint(ordered[i0], a);
1411 m_NavMesh->GetPoint(ordered[i1], b);
1412 vtkMath::Subtract(b, a, tangent);
1413 if (!Normalize3(tangent))
1414 {
1415 tangent[0] = 0.0;
1416 tangent[1] = 0.0;
1417 tangent[2] = 1.0;
1418 }
1419 };
1420
1421 double tangent[3] = { 0.0, 0.0, 1.0 };
1422 SampleTangent(0, tangent);
1423
1424 double localRadius = 12.0;
1425 double prevCenter[3];
1426 Copy3(start_pos, prevCenter);
1427 RecenterInLumen(
1428 m_CellLocator,
1429 start_pos,
1430 tangent,
1431 AdaptiveMaxRay(localRadius),
1432 AdaptiveMaxShift(localRadius),
1433 prevCenter,
1434 &localRadius);
1435 centers.push_back({ prevCenter[0], prevCenter[1], prevCenter[2] });
1436
1437 constexpr double kSampleStep = 1.6;
1438 double lastWall[3];
1439 m_NavMesh->GetPoint(ordered.front(), lastWall);
1440
1441 for (size_t n = 1; n < ordered.size(); ++n)
1442 {
1443 double p[3];
1444 m_NavMesh->GetPoint(ordered[n], p);
1445 const double edgeLen = std::sqrt(vtkMath::Distance2BetweenPoints(lastWall, p));
1446 if (edgeLen < 1e-6)
1447 {
1448 continue;
1449 }
1450
1451 SampleTangent(n, tangent);
1452 const int subCount = std::max(1, static_cast<int>(std::ceil(edgeLen / kSampleStep)));
1453 double edgeStart[3];
1454 Copy3(lastWall, edgeStart);
1455
1456 for (int s = 1; s <= subCount; ++s)
1457 {
1458 const double t = static_cast<double>(s) / static_cast<double>(subCount);
1459 double q[3] = {
1460 edgeStart[0] + t * (p[0] - edgeStart[0]),
1461 edgeStart[1] + t * (p[1] - edgeStart[1]),
1462 edgeStart[2] + t * (p[2] - edgeStart[2])
1463 };
1464
1465 double disp[3] = {
1466 q[0] - lastWall[0],
1467 q[1] - lastWall[1],
1468 q[2] - lastWall[2]
1469 };
1470 double proposed[3] = {
1471 prevCenter[0] + disp[0],
1472 prevCenter[1] + disp[1],
1473 prevCenter[2] + disp[2]
1474 };
1475
1476 double stepTan[3] = { disp[0], disp[1], disp[2] };
1477 if (!Normalize3(stepTan))
1478 {
1479 Copy3(tangent, stepTan);
1480 }
1481
1482 double centered[3];
1483 double radius = localRadius;
1484 if (!RecenterInLumen(
1485 m_CellLocator,
1486 proposed,
1487 stepTan,
1488 AdaptiveMaxRay(localRadius),
1489 AdaptiveMaxShift(localRadius),
1490 centered,
1491 &radius))
1492 {
1493 Copy3(proposed, centered);
1494 }
1495
1496 double corr[3] = {
1497 centered[0] - proposed[0],
1498 centered[1] - proposed[1],
1499 centered[2] - proposed[2]
1500 };
1501 SubtractTangent(corr, stepTan);
1502 centered[0] = proposed[0] + corr[0];
1503 centered[1] = proposed[1] + corr[1];
1504 centered[2] = proposed[2] + corr[2];
1505
1506 if (radius < localRadius)
1507 {
1508 localRadius = 0.25 * localRadius + 0.75 * radius;
1509 }
1510 else
1511 {
1512 localRadius = 0.75 * localRadius + 0.25 * radius;
1513 }
1514
1515 Copy3(q, lastWall);
1516 Copy3(centered, prevCenter);
1517
1518 const double moved = std::sqrt(vtkMath::Distance2BetweenPoints(
1519 centered, centers.back().data()));
1520 if (moved < 0.55)
1521 {
1522 continue;
1523 }
1524 centers.push_back({ centered[0], centered[1], centered[2] });
1525 }
1526 }
1527
1528 if (!centers.empty())
1529 {
1530 const double tail = std::sqrt(vtkMath::Distance2BetweenPoints(prevCenter, centers.back().data()));
1531 if (tail > 0.4)
1532 {

1577std::vector<TrackPoint> BronchoEngine::ComputeRoute(double start_pos[3], double target_pos[3]) {
1578 std::vector<TrackPoint> track;
1579
1580 // The voxelised lumen is authoritative: it cannot produce a point outside
1581 // the airway. Surface routing only covers models that fail to voxelise.
1582 std::vector<std::array<double, 3>> centers;
1583 if (!ComputeLumenCenters(start_pos, target_pos, centers))
1584 {
1585 centers = ComputeSurfaceCenters(start_pos, target_pos);
1586 }
1587
1588 if (centers.size() < 2)
1589 {
1590 TrackPoint tp;
1591 Copy3(centers.empty() ? start_pos : centers.front().data(), tp.pos);
1592 tp.focal[0] = target_pos[0];
1593 tp.focal[1] = target_pos[1];
1594 tp.focal[2] = target_pos[2];
1595 tp.up[0] = 0.0;
1596 tp.up[1] = 0.0;
1597 tp.up[2] = 1.0;
1598 track.push_back(tp);
1599 AimTrackAtTarget(track, target_pos);
1600 return track;
1601 }
1602
1603 std::vector<double> prefix(centers.size(), 0.0);
1604 for (size_t s = 1; s < centers.size(); ++s)
1605 {
1606 prefix[s] = prefix[s - 1] + std::sqrt(
1607 vtkMath::Distance2BetweenPoints(centers[s - 1].data(), centers[s].data()));
1608 }
1609
1610 const double totalLen = prefix.back();
1611 if (totalLen < 1e-3)
1612 {
1613 TrackPoint tp;
1614 Copy3(centers.front().data(), tp.pos);
1615 Copy3(target_pos, tp.focal);
1616 track.push_back(tp);
1617 AimTrackAtTarget(track, target_pos);
1618 return track;
1619 }
1620
1621 const int sampleCount = std::max(400, static_cast<int>(totalLen * 8.0));
1622 const double lookWindow = 20.0;
1623 const double lookDist = 6.0;
1624
1625 auto PointAtArc = [&](double arc, double out[3])
1626 {
1627 arc = std::max(0.0, std::min(totalLen, arc));
1628 size_t seg = 0;
1629 while (seg + 1 < prefix.size() && prefix[seg + 1] < arc)
1630 {
1631 ++seg;
1632 }
1633 if (seg + 1 >= centers.size())
1634 {
1635 Copy3(centers.back().data(), out);
1636 return;
1637 }
1638 const double segLen = prefix[seg + 1] - prefix[seg];
1639 const double t = (segLen > 1e-6) ? (arc - prefix[seg]) / segLen : 0.0;
1640 out[0] = centers[seg][0] + t * (centers[seg + 1][0] - centers[seg][0]);
1641 out[1] = centers[seg][1] + t * (centers[seg + 1][1] - centers[seg][1]);
1642 out[2] = centers[seg][2] + t * (centers[seg + 1][2] - centers[seg][2]);
1643 };
1644
1645 track.resize(static_cast<size_t>(sampleCount) + 1);
1646 std::vector<std::array<double, 3>> headings(track.size());
1647
1648 for (int j = 0; j <= sampleCount; ++j)
1649 {
1650 const double arc = totalLen * static_cast<double>(j) / static_cast<double>(sampleCount);
1651 TrackPoint& tp = track[static_cast<size_t>(j)];
1652 PointAtArc(arc, tp.pos);
1653
1654 double ahead[3];
1655 double behind[3];
1656 PointAtArc(std::min(totalLen, arc + lookWindow), ahead);
1657 PointAtArc(std::max(0.0, arc - lookWindow * 0.35), behind);
1658 double fwd[3] = {
1659 ahead[0] - behind[0],
1660 ahead[1] - behind[1],
1661 ahead[2] - behind[2]
1662 };
1663 if (!Normalize3(fwd))
1664 {
1665 fwd[0] = target_pos[0] - tp.pos[0];
1666 fwd[1] = target_pos[1] - tp.pos[1];
1667 fwd[2] = target_pos[2] - tp.pos[2];
1668 Normalize3(fwd);
1669 }
1670 headings[static_cast<size_t>(j)] = { fwd[0], fwd[1], fwd[2] };
1671 }
1672
1673 const double samplesPerMm = static_cast<double>(sampleCount) / totalLen;
1674 SmoothDirections(headings, std::max(1, static_cast<int>(10.0 * samplesPerMm)), 3);
1675
1676 for (size_t j = 0; j < track.size(); ++j)
1677 {
1678 track[j].focal[0] = track[j].pos[0] + headings[j][0] * lookDist;
1679 track[j].focal[1] = track[j].pos[1] + headings[j][1] * lookDist;
1680 track[j].focal[2] = track[j].pos[2] + headings[j][2] * lookDist;
1681 }
1682
1683 ApplyStableViewUp(track);
1684 return track;
1685}

1690void BronchoEngine::TrimRouteBeforeTarget(std::vector<TrackPoint>& track, const double target[3], double stopDist)
1691{
1692 if (track.size() < 2)
1693 {
1694 return;
1695 }
1696
1697 const double stop = std::max(0.0, stopDist);
1698 int nearest = static_cast<int>(track.size()) - 1;
1699 double nearestDist2 = vtkMath::Distance2BetweenPoints(track[nearest].pos, target);
1700 for (int i = 0; i < static_cast<int>(track.size()); ++i)
1701 {
1702 const double d2 = vtkMath::Distance2BetweenPoints(track[i].pos, target);
1703 if (d2 < nearestDist2)
1704 {
1705 nearestDist2 = d2;
1706 nearest = i;
1707 }
1708 }
1709
1710 double travelled = 0.0;
1711 int cut = nearest;
1712 while (cut > 0 && travelled < stop)
1713 {
1714 travelled += std::sqrt(
1715 vtkMath::Distance2BetweenPoints(track[cut].pos, track[cut - 1].pos));
1716 --cut;
1717 }
1718 track.resize(static_cast<size_t>(cut) + 1);
1719}

1724void BronchoEngine::AimTrackAtTarget(std::vector<TrackPoint>& track, const double target[3])
1725{
1726 if (track.empty())
1727 {
1728 return;
1729 }
1730
1731 const size_t n = track.size();
1732 std::vector<double> prefix(n, 0.0);
1733 for (size_t i = 1; i < n; ++i)
1734 {
1735 prefix[i] = prefix[i - 1] + std::sqrt(
1736 vtkMath::Distance2BetweenPoints(track[i - 1].pos, track[i].pos));
1737 }
1738 const double total = prefix.back();
1739
1740 constexpr double kBlendDist = 18.0;
1741 constexpr double kLookDist = 6.0;
1742
1743 std::vector<std::array<double, 3>> headings(n);
1744 for (size_t i = 0; i < n; ++i)
1745 {
1746 double fwd[3] = {
1747 track[i].focal[0] - track[i].pos[0],
1748 track[i].focal[1] - track[i].pos[1],
1749 track[i].focal[2] - track[i].pos[2]
1750 };
1751 if (!Normalize3(fwd))
1752 {
1753 fwd[0] = 0.0;
1754 fwd[1] = 0.0;
1755 fwd[2] = -1.0;
1756 }
1757 headings[i] = { fwd[0], fwd[1], fwd[2] };
1758 }
1759
1760 if (total > 1e-6 && n > 2)
1761 {
1762 const double samplesPerMm = static_cast<double>(n - 1) / total;
1763 SmoothDirections(headings, std::max(1, static_cast<int>(10.0 * samplesPerMm)), 3);
1764 }
1765
1766 for (size_t i = 0; i < n; ++i)
1767 {
1768 double aim[3] = {
1769 target[0] - track[i].pos[0],
1770 target[1] - track[i].pos[1],
1771 target[2] - track[i].pos[2]
1772 };
1773 const bool haveAim = Normalize3(aim);
1774 double fwd[3] = { headings[i][0], headings[i][1], headings[i][2] };
1775
1776 if (haveAim && total > 1e-6)
1777 {
1778 const double remaining = total - prefix[i];
1779 const double t = 1.0 - std::min(1.0, remaining / kBlendDist);
1780 const double w = t * t * (3.0 - 2.0 * t);
1781 double blended[3] = {
1782 fwd[0] * (1.0 - w) + aim[0] * w,
1783 fwd[1] * (1.0 - w) + aim[1] * w,
1784 fwd[2] * (1.0 - w) + aim[2] * w
1785 };
1786 if (Normalize3(blended))
1787 {
1788 Copy3(blended, fwd);
1789 }
1790 }
1791 else if (haveAim && !Normalize3(fwd))
1792 {
1793 Copy3(aim, fwd);
1794 }
1795
1796 headings[i] = { fwd[0], fwd[1], fwd[2] };
1797 }
1798
1799 for (size_t i = 0; i < n; ++i)
1800 {
1801 track[i].focal[0] = track[i].pos[0] + headings[i][0] * kLookDist;
1802 track[i].focal[1] = track[i].pos[1] + headings[i][1] * kLookDist;
1803 track[i].focal[2] = track[i].pos[2] + headings[i][2] * kLookDist;
1804 }
1805
1806 // The last pose looks at the target itself. Stop-distance keeps this
1807 // several millimetres away, so the bearing is stable.
1808 Copy3(target, track.back().focal);
1809
1810 ApplyStableViewUp(track);
1811}
