24bool HasStl(const fs::path& folder)
25{
26 std::error_code ec;
27 if (!fs::is_directory(folder, ec))
28 {
29 return false;
30 }
31 for (const auto& entry : fs::directory_iterator(folder, ec))

34 if (name.size() >= 4)

36 const std::string ext = name.substr(name.size() - 4);

64std::string FindAirwayFolder()
65{
66 for (const auto& root : CandidateRoots("airway_segments"))
67 {
68 if (!fs::is_directory(root))
69 {
70 continue;
71 }
72 if (HasStl(root))
73 {
74 return root.string();
75 }
76
77 std::vector<fs::path> children;
78 for (const auto& entry : fs::directory_iterator(root))
79 {
80 if (entry.is_directory())
81 {
82 children.push_back(entry.path());
83 }
84 }
85 std::sort(children.begin(), children.end());

101std::string FindCTVolume()
102{
103 for (const auto& root : CandidateRoots("CT_Data"))
104 {
105 if (!fs::is_directory(root))
106 {
107 continue;
108 }
109
110 std::vector<fs::path> files;
111 for (const auto& entry : fs::recursive_directory_iterator(root))
112 {
113 if (!entry.is_regular_file())
114 {
115 continue;
116 }
117 const std::string name = entry.path().filename().string();
118 if (name.size() >= 4 && name.substr(name.size() - 4) == ".nii")
119 {
120 files.push_back(entry.path());
121 }
122 else if (name.size() >= 7 && name.substr(name.size() - 7) == ".nii.gz")
123 {
124 files.push_back(entry.path());
125 }
126 }
127 std::sort(files.begin(), files.end());
128
129 for (const auto& candidate : files)
130 {
131 auto reader = vtkSmartPointer<vtkNIFTIImageReader>::New();
132 if (reader->CanReadFile(candidate.string().c_str()) != 0)
133 {
134 return candidate.string();
135 }
136 std::cout << "[CT] Skipping unreadable volume: "
137 << candidate.filename().string() << "\n";
138 }
139 }
140 return {};
