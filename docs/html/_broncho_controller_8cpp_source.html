52void TrimRouteToEndAt(std::vector<TrackPoint>& route, double retractMm)
53{
54 double remaining = std::max(0.0, retractMm);
55 while (route.size() > 1 && remaining > 1e-6)
56 {
57 const double seg = std::sqrt(
58 vtkMath::Distance2BetweenPoints(route.back().pos, route[route.size() - 2].pos));
59 if (seg <= remaining)
60 {
61 remaining -= seg;
62 route.pop_back();
63 continue;
64 }
65
66 const double t = remaining / seg;
67 TrackPoint& tip = route.back();
68 const TrackPoint& prev = route[route.size() - 2];
69 for (int i = 0; i < 3; ++i)
70 {
71 tip.pos[i] += t * (prev.pos[i] - tip.pos[i]);
72 }
73 break;
74 }
75}

80void ApplyLevelHorizon(vtkCamera* cam)
81{
82 if (cam == nullptr)
83 {
84 return;
85 }
86
87 double pos[3], focal[3], fwd[3];
88 cam->GetPosition(pos);
89 cam->GetFocalPoint(focal);
90 vtkMath::Subtract(focal, pos, fwd);
91 if (vtkMath::Normalize(fwd) < 1e-9)
92 {
93 return;
94 }
95
96 double worldZ[3] = { 0.0, 0.0, 1.0 };
97 double right[3];
98 vtkMath::Cross(fwd, worldZ, right);
99 if (vtkMath::Norm(right) < 0.12)
100 {
101 cam->OrthogonalizeViewUp();
102 return;
103 }
104
105 const double along = vtkMath::Dot(worldZ, fwd);
106 double up[3] = {
107 worldZ[0] - along * fwd[0],
108 worldZ[1] - along * fwd[1],
109 worldZ[2] - along * fwd[2]
110 };
111 if (vtkMath::Normalize(up) < 1e-9)
112 {
113 cam->OrthogonalizeViewUp();
114 return;
115 }
116 cam->SetViewUp(up);
117}

184std::string FindNiftiInTree(const std::filesystem::path& root)
185{
186 std::error_code ec;
187 if (!std::filesystem::is_directory(root, ec))
188 {
189 return {};
190 }
191
192 std::vector<std::filesystem::path> files;
193 for (const auto& entry : std::filesystem::recursive_directory_iterator(root, ec))
194 {
195 if (entry.is_regular_file() && IsNiftiFile(entry.path()))
196 {
197 files.push_back(entry.path());
198 }
199 }
200 std::sort(files.begin(), files.end());
201 if (files.empty())
202 {
203 return {};
204 }
205 return files.front().string();
206}

326void BronchoController::JumpToAirwayPoint(const double destIn[3], bool isTarget)
327{
328 m_IsPlaying = false;
329 double dest[3] = { destIn[0], destIn[1], destIn[2] };
330
331 double stop_dists[] = { 6.0, 12.0, 20.0 };
332 double stop_dist = stop_dists[m_StopDistanceMode];
333
334 double start[3] = { 0.0, 0.0, 0.0 };
335 bool has_trachea = false;
336 double start_z_third = 0.0;
337
338 if (m_Engine->GetTracheaActor())
339 {
340 double tb[6];
341 m_Engine->GetTracheaActor()->GetBounds(tb);
342
343 if (m_Engine->GetSegments().size() == 1)
344 {
345 start[0] = m_Engine->GetTracheaStartPos()[0];
346 start[1] = m_Engine->GetTracheaStartPos()[1];
347 start[2] = m_Engine->GetTracheaStartPos()[2];
348 }
349 else
350 {
351 start[0] = (tb[0] + tb[1]) / 2.0;
352 start[1] = (tb[2] + tb[3]) / 2.0;
353 start[2] = tb[5] - 5.0;
354 start_z_third = tb[5] - ((tb[5] - tb[4]) / 3.0);
355 }
356
357 has_trachea = true;
358 }
359 else
360 {
361 m_View->GetRightRenderer()->GetActiveCamera()->GetPosition(start);
362 }
363
364 if (!isTarget)
365 {
366 if (IsNearTracheaEntry(m_Engine.get(), dest))
367 {
368 // A wall pick at the inlet makes a short radial route whose heading
369 // yaws the camera. Snap back to the same look-down pose as Reset.
370 ApplyEndoscopeEntryPose(m_Engine.get(), m_View.get());
371 m_ActiveRoute.clear();
372 m_TrackIndex = 0;
373 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
374 ForceUpdateMinimap();
375 return;
376 }
377
378 m_ActiveRoute = m_Engine->ComputeRoute(start, dest);
379 // Stop short of the wall pick, but keep the path-tangent heading so
380 // the camera faces down the airway instead of at the clicked surface.
381 BronchoEngine::TrimRouteBeforeTarget(m_ActiveRoute, dest, stop_dist);
382
383 m_TrackIndex = m_ActiveRoute.empty() ? 0 : (int)m_ActiveRoute.size() - 1;
384
385 auto cam = m_View->GetRightRenderer()->GetActiveCamera();
386 if (!m_ActiveRoute.empty())
387 {
388 ApplyRoutePose(cam, m_ActiveRoute[m_TrackIndex]);
389 }
390 else
391 {
392 cam->OrthogonalizeViewUp();
393 }
394
395 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
396
397 }
398 else
399 {
400 // --- Save the target location to our plan! ---
401 m_ActiveTargetPos[0] = dest[0];
402 m_ActiveTargetPos[1] = dest[1];
403 m_ActiveTargetPos[2] = dest[2];
404
405 m_View->GetMarkerActor()->SetPosition(dest);
406 m_View->GetMarkerActor()->SetVisibility(true);
407 m_View->GetMarkerActorRight()->SetPosition(dest);
408 m_View->GetMarkerActorRight()->SetVisibility(true);
409
410 auto cam = m_View->GetRightRenderer()->GetActiveCamera();
411
412 m_ActiveRoute = m_Engine->ComputeRoute(start, dest);
413 FinalizeRouteToTarget(m_ActiveRoute, dest, stop_dist);
414
415 m_TrackIndex = 0;
416
417 if (has_trachea && !m_ActiveRoute.empty())
418 {
419 if (m_Engine->GetSegments().size() > 1)
420 {
421 for (int i = 0; i < (int)m_ActiveRoute.size(); ++i)
422 {
423 if (m_ActiveRoute[i].pos[2] <= start_z_third)
424 {
425 m_TrackIndex = i;
426 break;
427 }
428 }
429 }
430 }
431
432 if (!m_ActiveRoute.empty())
433 {
434 ApplyRoutePose(cam, m_ActiveRoute[m_TrackIndex]);
435 }
436
437 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
438 }
439
440 ForceUpdateMinimap();
441}

446void BronchoController::FreeFly(double distance)
447{
448 m_IsPlaying = false;
449 auto cam = m_View->GetRightRenderer()->GetActiveCamera();
450 double pos[3], focal[3];
451 cam->GetPosition(pos);
452 cam->GetFocalPoint(focal);
453
454 double dir_vec[3];
455 vtkMath::Subtract(focal, pos, dir_vec);
456 double length = vtkMath::Norm(dir_vec);
457 if (length == 0) return;
458
459 double speed_factor = (m_SpeedMode == 0) ? 0.5 : (m_SpeedMode == 1) ? 1.0 : 2.0;
460 double actual_distance = distance * speed_factor;
461 const double look[3] = { dir_vec[0] / length, dir_vec[1] / length, dir_vec[2] / length };
462
463 if (!m_ActiveRoute.empty() && m_TrackIndex < (int)m_ActiveRoute.size() - 1)
464 {
465 m_ActiveRoute.resize(m_TrackIndex + 1);
466 }
467
468 double new_pos[3], new_focal[3];
469 double up[3] = { 0.0, 1.0, 0.0 };
470 cam->GetViewUp(up);
471
472 if (actual_distance < 0.0 && !m_ActiveRoute.empty())
473 {
474 // Retract along the catheter, not the current look vector, so down
475 // cannot fly the camera into the tube after a look or branch turn.
476 TrimRouteToEndAt(m_ActiveRoute, -actual_distance);
477 for (int i = 0; i < 3; ++i)
478 {
479 new_pos[i] = m_ActiveRoute.back().pos[i];
480 new_focal[i] = new_pos[i] + look[i] * length;
481 }
482 if (!m_Engine->IsMoveValid(new_pos))
483 {
484 ForceUpdateMinimap();
485 return;
486 }
487
488 cam->SetPosition(new_pos);
489 cam->SetFocalPoint(new_focal);
490 TrackPoint& tip = m_ActiveRoute.back();
491 tip.pos[0] = new_pos[0]; tip.pos[1] = new_pos[1]; tip.pos[2] = new_pos[2];
492 tip.focal[0] = new_focal[0]; tip.focal[1] = new_focal[1]; tip.focal[2] = new_focal[2];
493 tip.up[0] = up[0]; tip.up[1] = up[1]; tip.up[2] = up[2];
494 m_TrackIndex = static_cast<int>(m_ActiveRoute.size()) - 1;
495 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
496 ForceUpdateMinimap();
497 return;
498 }
499
500 for (int i = 0; i < 3; ++i)
501 {
502 new_pos[i] = pos[i] + look[i] * actual_distance;
503 new_focal[i] = focal[i] + look[i] * actual_distance;
504 }
505
506 if (m_Engine->IsMoveValid(new_pos))
507 {
508 cam->SetPosition(new_pos);
509 cam->SetFocalPoint(new_focal);
510
511 if (m_ActiveRoute.empty())
512 {
513 double start[3] = { 0.0, 0.0, 0.0 };
514 if (m_Engine->GetTracheaActor())
515 {
516 if (m_Engine->GetSegments().size() == 1)
517 {
518 start[0] = m_Engine->GetTracheaStartPos()[0];
519 start[1] = m_Engine->GetTracheaStartPos()[1];
520 start[2] = m_Engine->GetTracheaStartPos()[2];
521 }
522 else
523 {
524 double tb[6];
525 m_Engine->GetTracheaActor()->GetBounds(tb);
526 start[0] = (tb[0] + tb[1]) / 2.0;
527 start[1] = (tb[2] + tb[3]) / 2.0;
528 start[2] = tb[5] - 5.0;
529 }
530 }
531 else
532 {
533 start[0] = pos[0]; start[1] = pos[1]; start[2] = pos[2];
534 }
535 m_ActiveRoute = m_Engine->ComputeRoute(start, new_pos);
536 m_TrackIndex = m_ActiveRoute.empty() ? 0 : (int)m_ActiveRoute.size() - 1;
537 }
538 else
539 {
540 TrackPoint tp;
541 tp.pos[0] = new_pos[0]; tp.pos[1] = new_pos[1]; tp.pos[2] = new_pos[2];
542 tp.focal[0] = new_focal[0]; tp.focal[1] = new_focal[1]; tp.focal[2] = new_focal[2];
543 tp.up[0] = up[0]; tp.up[1] = up[1]; tp.up[2] = up[2];
544 m_ActiveRoute.push_back(tp);
545 m_TrackIndex = (int)m_ActiveRoute.size() - 1;
546 }
547 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
548 }
549 ForceUpdateMinimap();
550}

626void BronchoController::ForceUpdateMinimap()
627{
628 vtkCamera* cam = m_View->GetRightRenderer()->GetActiveCamera();
629 double pos[3], focal[3];
630 cam->GetPosition(pos);
631 cam->GetFocalPoint(focal);
632
633 m_View->GetHighlightSphere()->SetCenter(pos);
634 m_View->GetCTSlices().Update(pos);
635
636 double dir[3] = { focal[0] - pos[0], focal[1] - pos[1], focal[2] - pos[2] };
637 double length = vtkMath::Norm(dir);
638 if (length > 0)
639 {
640 double rev_dir[3] = { -(dir[0] / length), -(dir[1] / length), -(dir[2] / length) };
641 m_View->GetFOVSource()->SetDirection(rev_dir);
642 double height = m_View->GetFOVSource()->GetHeight();
643 double conePos[3] = { pos[0] - rev_dir[0] * (height / 2.0), pos[1] - rev_dir[1] * (height / 2.0), pos[2] - rev_dir[2] * (height / 2.0) };
644 m_View->GetFOVActor()->SetPosition(conePos);
645 }
646
647 vtkActor* best_actor = nullptr;
648 double best_score = VTK_DOUBLE_MAX;
649 for (const auto& seg : m_Engine->GetSegments())
650 {
651 double dist2 = vtkMath::Distance2BetweenPoints(pos, seg.center);
652 if (pos[0] >= seg.bounds[0] && pos[0] <= seg.bounds[1] &&
653 pos[1] >= seg.bounds[2] && pos[1] <= seg.bounds[3] &&
654 pos[2] >= seg.bounds[4] && pos[2] <= seg.bounds[5])
655 {
656 dist2 -= 1000000.0;
657 }
658 if (dist2 < best_score)
659 {
660 best_score = dist2; best_actor = seg.actor;
661 }
662 }
663
664 if (best_actor != m_View->GetActiveSegment())
665 {
666 if (m_View->GetActiveSegment())
667 {
668 m_View->GetActiveSegment()->GetProperty()->SetColor(0.9, 0.7, 0.7);
669 m_View->GetActiveSegment()->GetProperty()->SetAmbient(0.0);
670 }
671 if (best_actor)
672 {
673 best_actor->GetProperty()->SetColor(0.0, 0.5, 1.0);
674 best_actor->GetProperty()->SetAmbient(0.6);
675 }
676 m_View->SetActiveSegment(best_actor);
677 }
678}

743void BronchoController::Tick()
744{
745 if (!m_IsPlaying) return;
746
747 if (!m_ActiveRoute.empty() && m_TrackIndex < (int)m_ActiveRoute.size() - 1)
748 {
749 int step = (m_SpeedMode == 0) ? 2 : (m_SpeedMode == 1) ? 5 : 10;
750 m_TrackIndex += step;
751
752 if (m_TrackIndex >= (int)m_ActiveRoute.size())
753 {
754 m_TrackIndex = (int)m_ActiveRoute.size() - 1;
755 }
756
757 vtkCamera* cam = m_View->GetRightRenderer()->GetActiveCamera();
758 ApplyRoutePose(cam, m_ActiveRoute[m_TrackIndex]);
759 if (m_TrackIndex >= (int)m_ActiveRoute.size() - 1)
760 {
761 if (m_View->GetMarkerActor() != nullptr && m_View->GetMarkerActor()->GetVisibility())
762 {
763 cam->SetFocalPoint(m_ActiveTargetPos);
764 cam->OrthogonalizeViewUp();
765 }
766 m_IsPlaying = false;
767 }
768 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
769 UpdateMinimap();
770 }
771 else
772 {
773 m_IsPlaying = false;
774 if (!m_ActiveRoute.empty() && m_View != nullptr)
775 {
776 vtkCamera* cam = m_View->GetRightRenderer()->GetActiveCamera();
777 ApplyRoutePose(cam, m_ActiveRoute.back());
778 if (m_View->GetMarkerActor() != nullptr && m_View->GetMarkerActor()->GetVisibility())
779 {
780 cam->SetFocalPoint(m_ActiveTargetPos);
781 cam->OrthogonalizeViewUp();
782 }
783 }
784 std::cout << "[CONTROLLER] Destination Reached." << std::endl;
785 }

820bool BronchoController::LoadPlan(const std::string& filePath,
821 const std::function<void(const std::string&, int)>& onProgress)
822{
823 auto report = [&](const std::string& step, int percent)
824 {
825 if (onProgress)
826 {
827 onProgress(step, percent);
828 }
829 };
830
831 report("Reading plan", 4);
832 if (!m_ActivePlan.LoadFromFile(filePath))
833 {
834 return false;
835 }
836
837 m_TrackIndex = 0;
838 m_IsPlaying = false;
839
840 if (!m_ActivePlan.stlFilePath.empty())
841 {
842 report("Loading airways", 12);
843 LoadNewModel(m_ActivePlan.stlFilePath, onProgress);
844 }
845
846 const std::string ctPath = ResolvePlanCTPath(m_ActivePlan, filePath);
847 if (!ctPath.empty())
848 {
849 LoadCT(ctPath, onProgress);
850 }
851
852 report("Updating scene", 94);
853 // Redraw route and target based on loaded data
854 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
855
856 // Only set marker position if it's not strictly {0,0,0} to avoid artifacts
857 if (m_ActiveTargetPos[0] != 0.0 || m_ActiveTargetPos[1] != 0.0 || m_ActiveTargetPos[2] != 0.0)
858 {
859 m_View->GetMarkerActor()->SetPosition(m_ActiveTargetPos);
860 m_View->GetMarkerActor()->SetVisibility(true);
861 m_View->GetMarkerActorRight()->SetPosition(m_ActiveTargetPos);
862 m_View->GetMarkerActorRight()->SetVisibility(true);
863 }
864 else
865 {
866 m_View->GetMarkerActor()->SetVisibility(false);
867 m_View->GetMarkerActorRight()->SetVisibility(false);
868 }
869
870 UpdateSavedTargetVisuals();
871 report("Done", 100);
872
873 return true;
874}

916void BronchoController::UpdateSavedTargetVisuals()
917{
918 if (m_View == nullptr || m_View->GetLeftRenderer() == nullptr || m_View->GetRightRenderer() == nullptr)
919 {
920 return;
921 }
922
923 for (auto& actor : m_SavedTargetActorsLeft)
924 {
925 m_View->GetLeftRenderer()->RemoveActor(actor);
926 }
927 for (auto& actor : m_SavedTargetActorsRight)
928 {
929 m_View->GetRightRenderer()->RemoveActor(actor);
930 }
931
932 m_SavedTargetActorsLeft.clear();
933 m_SavedTargetActorsRight.clear();
934
935 for (const auto& target : m_ActivePlan.targets)
936 {
937 auto sphere = vtkSmartPointer<vtkSphereSource>::New();
938 sphere->SetRadius(3.0);
939 sphere->SetCenter(target.location);
940 sphere->Update();
941
942 auto mapper = vtkSmartPointer<vtkPolyDataMapper>::New();
943 mapper->SetInputData(sphere->GetOutput());
944
945 auto actorLeft = vtkSmartPointer<vtkActor>::New();
946 actorLeft->SetMapper(mapper);
947 actorLeft->GetProperty()->SetColor(0.2, 0.8, 0.2);
948
949 auto actorRight = vtkSmartPointer<vtkActor>::New();
950 actorRight->SetMapper(mapper);
951 actorRight->GetProperty()->SetColor(0.2, 0.8, 0.2);
952
953 m_View->GetLeftRenderer()->AddActor(actorLeft);
954 m_View->GetRightRenderer()->AddActor(actorRight);
955
956 m_SavedTargetActorsLeft.push_back(actorLeft);
957 m_SavedTargetActorsRight.push_back(actorRight);
958 }
959}

966void BronchoController::LoadSavedTarget(size_t index)
967{
968 if (index < m_ActivePlan.targets.size())
969 {
970 const auto& target = m_ActivePlan.targets[index];
971 m_ActiveRoute = target.route;
972 BronchoEngine::AimTrackAtTarget(m_ActiveRoute, target.location);
973
974 m_ActiveTargetPos[0] = target.location[0];
975 m_ActiveTargetPos[1] = target.location[1];
976 m_ActiveTargetPos[2] = target.location[2];
977
978 m_TrackIndex = 0;
979
980 if (m_View != nullptr)
981 {
982 m_View->GetMarkerActor()->SetPosition(m_ActiveTargetPos);
983 m_View->GetMarkerActor()->SetVisibility(true);
984 m_View->GetMarkerActorRight()->SetPosition(m_ActiveTargetPos);
985 m_View->GetMarkerActorRight()->SetVisibility(true);
986
987 m_View->UpdateBreadcrumbs(m_ActiveRoute, m_TrackIndex);
988
989 auto cam = m_View->GetRightRenderer()->GetActiveCamera();
990 if (!m_ActiveRoute.empty())
991 {
992 ApplyRoutePose(cam, m_ActiveRoute[m_TrackIndex]);
993 }
994 }
995
996 ForceUpdateMinimap();
997 }
