75inline std::filesystem::path WriteBentTubeStl(
76 const std::filesystem::path& directory,
77 double ringRadius = 18.0,
78 double tubeRadius = 7.0)
79{
80 std::filesystem::create_directories(directory);
81
82 const int sampleCount = 80;
83 const double halfPi = std::acos(-1.0) * 0.5;
84 auto points = vtkSmartPointer<vtkPoints>::New();
85 auto line = vtkSmartPointer<vtkPolyLine>::New();
86 line->GetPointIds()->SetNumberOfIds(sampleCount + 1);
87 for (int i = 0; i <= sampleCount; ++i)
88 {
89 const double theta = halfPi * static_cast<double>(i) / static_cast<double>(sampleCount);
90 points->InsertNextPoint(ringRadius * std::cos(theta), 0.0, ringRadius * std::sin(theta));
91 line->GetPointIds()->SetId(i, i);
92 }
93
94 auto cells = vtkSmartPointer<vtkCellArray>::New();
95 cells->InsertNextCell(line);
96
97 auto centerline = vtkSmartPointer<vtkPolyData>::New();
98 centerline->SetPoints(points);
99 centerline->SetLines(cells);
100
101 auto tube = vtkSmartPointer<vtkTubeFilter>::New();
102 tube->SetInputData(centerline);
103 tube->SetRadius(tubeRadius);
104 tube->SetNumberOfSides(28);
105 tube->CappingOn();
106 tube->Update();
107
108 auto triangles = vtkSmartPointer<vtkTriangleFilter>::New();
109 triangles->SetInputData(tube->GetOutput());
110 triangles->Update();
111
112 const auto stlPath = directory / "Trachea.stl";
113 auto writer = vtkSmartPointer<vtkSTLWriter>::New();
114 writer->SetFileName(stlPath.string().c_str());
115 writer->SetInputConnection(triangles->GetOutputPort());
116 writer->SetFileTypeToBinary();
117 writer->Write();
118
119 return stlPath;
120}

299inline double DepthOutsideTube(const AirwayTube& tube, const double p[3])
300{
301 double axis[3] = {
302 tube.to[0] - tube.from[0],
303 tube.to[1] - tube.from[1],
304 tube.to[2] - tube.from[2]
305 };
306 const double axisLen2 = vtkMath::Dot(axis, axis);
307 double offset[3] = {
308 p[0] - tube.from[0],
309 p[1] - tube.from[1],
310 p[2] - tube.from[2]
311 };
312 double t = (axisLen2 > 1e-9) ? (vtkMath::Dot(offset, axis) / axisLen2) : 0.0;
313 t = std::max(0.0, std::min(1.0, t));
314
315 const double closest[3] = {
316 tube.from[0] + t * axis[0],
317 tube.from[1] + t * axis[1],
318 tube.from[2] + t * axis[2]
319 };
320 return std::sqrt(vtkMath::Distance2BetweenPoints(p, closest)) - tube.radius;
321}

375inline std::filesystem::path WriteMockAirNii(
376 const std::filesystem::path& directory,
377 const std::array<double, 3>& airCenter,
378 int dim = 40,
379 double airRadius = 6.0)
380{
381 std::filesystem::create_directories(directory);
382
383 auto image = vtkSmartPointer<vtkImageData>::New();
384 image->SetDimensions(dim, dim, dim);
385 image->SetSpacing(1.0, 1.0, 1.0);
386 image->SetOrigin(0.0, 0.0, 0.0);
387 image->AllocateScalars(VTK_SHORT, 1);
388
389 const double r2 = airRadius * airRadius;
390 for (int k = 0; k < dim; ++k)
391 {
392 for (int j = 0; j < dim; ++j)
393 {
394 for (int i = 0; i < dim; ++i)
395 {
396 const double dx = static_cast<double>(i) - airCenter[0];
397 const double dy = static_cast<double>(j) - airCenter[1];
398 const double dz = static_cast<double>(k) - airCenter[2];
399 *static_cast<short*>(image->GetScalarPointer(i, j, k)) =
400 ((dx * dx + dy * dy + dz * dz) <= r2) ? -1000 : -700;
401 }
402 }
403 }
404
405 const auto path = directory / "mock_airway.nii";
406 auto writer = vtkSmartPointer<vtkNIFTIImageWriter>::New();
407 writer->SetFileName(path.string().c_str());
408 writer->SetInputData(image);
409 writer->Write();
410 return path;
411}
