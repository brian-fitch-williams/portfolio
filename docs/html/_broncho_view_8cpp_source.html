289void BronchoView::UpdateBreadcrumbs(const std::vector<TrackPoint>& track, size_t currentIndex) {
290 auto bcPoints = vtkSmartPointer<vtkPoints>::New();
291
292 if (track.empty()) {
293 m_BreadcrumbPolyData->SetPoints(nullptr);
294 m_CatheterPolyData->SetPoints(nullptr);
295 m_CatheterPolyData->SetLines(nullptr);
296 return;
297 }
298
299 size_t step_size = 40;
300 for (size_t i = 0; i < track.size(); i += step_size) {
301 if (i >= currentIndex) bcPoints->InsertNextPoint(track[i].pos);
302 }
303 if (currentIndex < track.size() - 1) {
304 bcPoints->InsertNextPoint(track.back().pos);
305 }
306
307 m_BreadcrumbPolyData->SetPoints(bcPoints);
308 m_BreadcrumbPolyData->Modified();
309
310 constexpr double kCatheterRadius = 2.0;
311 constexpr double kTipClearance = kCatheterRadius + 1.5;
312
313 if (currentIndex > 0 && track.size() > 1) {
314 auto cathPts = vtkSmartPointer<vtkPoints>::New();
315
316 size_t last = currentIndex;
317 double backed = 0.0;
318 while (last > 0 && backed < kTipClearance) {
319 const double seg = std::sqrt(
320 vtkMath::Distance2BetweenPoints(track[last].pos, track[last - 1].pos));
321 if (backed + seg >= kTipClearance && seg > 1e-6) {
322 const double t = (kTipClearance - backed) / seg;
323 double tip[3] = {
324 track[last].pos[0] + t * (track[last - 1].pos[0] - track[last].pos[0]),
325 track[last].pos[1] + t * (track[last - 1].pos[1] - track[last].pos[1]),
326 track[last].pos[2] + t * (track[last - 1].pos[2] - track[last].pos[2])
327 };
328 for (size_t i = 0; i < last; ++i) {
329 cathPts->InsertNextPoint(track[i].pos);
330 }
331 cathPts->InsertNextPoint(tip);
332 last = static_cast<size_t>(-1);
333 break;
334 }
335 backed += seg;
336 --last;
337 }
338
339 if (last != static_cast<size_t>(-1)) {
340 for (size_t i = 0; i <= last; ++i) {
341 cathPts->InsertNextPoint(track[i].pos);
342 }
343 }
344
345 if (cathPts->GetNumberOfPoints() >= 2) {
346 auto polyLine = vtkSmartPointer<vtkPolyLine>::New();
347 polyLine->GetPointIds()->SetNumberOfIds(cathPts->GetNumberOfPoints());
348 for (vtkIdType i = 0; i < cathPts->GetNumberOfPoints(); ++i) {
349 polyLine->GetPointIds()->SetId(i, i);
350 }
351
352 auto cells = vtkSmartPointer<vtkCellArray>::New();
353 cells->InsertNextCell(polyLine);
354
355 m_CatheterPolyData->SetPoints(cathPts);
356 m_CatheterPolyData->SetLines(cells);
357 } else {
358 m_CatheterPolyData->SetPoints(nullptr);
359 m_CatheterPolyData->SetLines(nullptr);
360 }
361 } else {
362 m_CatheterPolyData->SetPoints(nullptr);
363 m_CatheterPolyData->SetLines(nullptr);
364 }
365 m_CatheterPolyData->Modified();
366}
