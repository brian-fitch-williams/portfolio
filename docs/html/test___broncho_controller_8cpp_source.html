214TEST_F(ControllerWithSceneTest, JumpBackToEntryKeepsLookingDown)
215{
216 m_controller.ResetToTrachea();
217 vtkCamera* cam = m_controller.GetEndoscopeRenderer()->GetActiveCamera();
218 ASSERT_NE(cam, nullptr);
219
220 double entryPos[3];
221 cam->GetPosition(entryPos);
222
223 for (int i = 0; i < 12; ++i)
224 {
225 m_controller.FreeFly(3.0);
226 }
227
228 double wallPick[3] = { entryPos[0] + 8.0, entryPos[1], entryPos[2] };
229 m_controller.JumpToAirwayPoint(wallPick, false);
230
231 double pos[3];
232 double focal[3];
233 double up[3];
234 cam->GetPosition(pos);
235 cam->GetFocalPoint(focal);
236 cam->GetViewUp(up);
237
238 EXPECT_LT(std::sqrt(vtkMath::Distance2BetweenPoints(pos, entryPos)), 2.0);
239
240 double fwd[3];
241 vtkMath::Subtract(focal, pos, fwd);
242 ASSERT_GT(vtkMath::Normalize(fwd), 1e-6);
243 EXPECT_LT(fwd[2], -0.8);
244 EXPECT_GT(up[1], 0.8);
245}
