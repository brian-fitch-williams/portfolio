497double AngleBetweenDegrees(const double a[3], const double b[3])
498{
499 double u[3] = { a[0], a[1], a[2] };
500 double v[3] = { b[0], b[1], b[2] };
501 if (vtkMath::Normalize(u) < 1e-9 || vtkMath::Normalize(v) < 1e-9)
502 {
503 return 0.0;
504 }
505 const double dot = std::max(-1.0, std::min(1.0, vtkMath::Dot(u, v)));
506 return vtkMath::DegreesFromRadians(std::acos(dot));
507}
